#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/wait.h>
#include <fcntl.h>

// ============================================================================
// 1. BASIC AST STRUCTURE DEFINITIONS
// ============================================================================

typedef enum {
    NODE_COMMAND,      // Simple command: ls -la
    NODE_PIPE,         // Pipe: cmd1 | cmd2
    NODE_REDIRECT_IN,  // Input redirect: cmd < file
    NODE_REDIRECT_OUT, // Output redirect: cmd > file
    NODE_REDIRECT_APPEND, // Append: cmd >> file
    NODE_SEQUENCE,     // Sequential: cmd1 ; cmd2
    NODE_AND,          // Logical AND: cmd1 && cmd2
    NODE_OR,           // Logical OR: cmd1 || cmd2
    NODE_BACKGROUND    // Background: cmd &
} NodeType;

typedef struct ASTNode {
    NodeType type;
    union {
        // For simple commands
        struct {
            char **argv;     // NULL-terminated array of arguments
            int argc;        // Argument count
        } command;
        
        // For binary operations (pipe, sequence, and, or)
        struct {
            struct ASTNode *left;
            struct ASTNode *right;
        } binary;
        
        // For redirections
        struct {
            struct ASTNode *command;
            char *filename;
            int flags;       // O_WRONLY, O_CREAT, etc.
            int fd;          // File descriptor to redirect (0=stdin, 1=stdout, 2=stderr)
        } redirect;
        
        // For background execution
        struct {
            struct ASTNode *command;
        } background;
    } data;
} ASTNode;

// Token structure for parsing
typedef struct {
    char *value;
    enum {
        TOKEN_WORD,
        TOKEN_PIPE,
        TOKEN_REDIRECT_IN,
        TOKEN_REDIRECT_OUT,
        TOKEN_REDIRECT_APPEND,
        TOKEN_SEMICOLON,
        TOKEN_AND,
        TOKEN_OR,
        TOKEN_BACKGROUND,
        TOKEN_EOF
    } type;
} Token;

// ============================================================================
// 2. AST NODE CONSTRUCTORS
// ============================================================================

ASTNode* create_command_node(char **argv, int argc) {
    ASTNode *node = malloc(sizeof(ASTNode));
    node->type = NODE_COMMAND;
    
    // Deep copy the arguments
    node->data.command.argv = malloc(sizeof(char*) * (argc + 1));
    node->data.command.argc = argc;
    
    for (int i = 0; i < argc; i++) {
        node->data.command.argv[i] = strdup(argv[i]);
    }
    node->data.command.argv[argc] = NULL;
    
    return node;
}

ASTNode* create_pipe_node(ASTNode *left, ASTNode *right) {
    ASTNode *node = malloc(sizeof(ASTNode));
    node->type = NODE_PIPE;
    node->data.binary.left = left;
    node->data.binary.right = right;
    return node;
}

ASTNode* create_redirect_node(NodeType type, ASTNode *command, char *filename) {
    ASTNode *node = malloc(sizeof(ASTNode));
    node->type = type;
    node->data.redirect.command = command;
    node->data.redirect.filename = strdup(filename);
    
    switch (type) {
        case NODE_REDIRECT_IN:
            node->data.redirect.fd = 0;  // stdin
            node->data.redirect.flags = O_RDONLY;
            break;
        case NODE_REDIRECT_OUT:
            node->data.redirect.fd = 1;  // stdout
            node->data.redirect.flags = O_WRONLY | O_CREAT | O_TRUNC;
            break;
        case NODE_REDIRECT_APPEND:
            node->data.redirect.fd = 1;  // stdout
            node->data.redirect.flags = O_WRONLY | O_CREAT | O_APPEND;
            break;
        default:
            break;
    }
    
    return node;
}

ASTNode* create_sequence_node(ASTNode *left, ASTNode *right) {
    ASTNode *node = malloc(sizeof(ASTNode));
    node->type = NODE_SEQUENCE;
    node->data.binary.left = left;
    node->data.binary.right = right;
    return node;
}

ASTNode* create_and_node(ASTNode *left, ASTNode *right) {
    ASTNode *node = malloc(sizeof(ASTNode));
    node->type = NODE_AND;
    node->data.binary.left = left;
    node->data.binary.right = right;
    return node;
}

ASTNode* create_or_node(ASTNode *left, ASTNode *right) {
    ASTNode *node = malloc(sizeof(ASTNode));
    node->type = NODE_OR;
    node->data.binary.left = left;
    node->data.binary.right = right;
    return node;
}

// ============================================================================
// 3. PARSER - Building AST from Tokens
// ============================================================================

typedef struct {
    Token *tokens;
    int position;
    int count;
} Parser;

ASTNode* parse_command(Parser *parser);
ASTNode* parse_pipeline(Parser *parser);
ASTNode* parse_sequence(Parser *parser);

// Parse a simple command (e.g., "ls -la /tmp")
ASTNode* parse_simple_command(Parser *parser) {
    char *argv[256];
    int argc = 0;
    
    // Collect all consecutive WORD tokens
    while (parser->position < parser->count && 
           parser->tokens[parser->position].type == TOKEN_WORD) {
        argv[argc++] = parser->tokens[parser->position].value;
        parser->position++;
    }
    
    if (argc == 0) {
        return NULL;
    }
    
    ASTNode *command = create_command_node(argv, argc);
    
    // Check for redirections
    while (parser->position < parser->count) {
        Token *current = &parser->tokens[parser->position];
        
        if (current->type == TOKEN_REDIRECT_IN) {
            parser->position++;
            if (parser->position < parser->count && 
                parser->tokens[parser->position].type == TOKEN_WORD) {
                command = create_redirect_node(NODE_REDIRECT_IN, command, 
                                              parser->tokens[parser->position].value);
                parser->position++;
            }
        } else if (current->type == TOKEN_REDIRECT_OUT) {
            parser->position++;
            if (parser->position < parser->count && 
                parser->tokens[parser->position].type == TOKEN_WORD) {
                command = create_redirect_node(NODE_REDIRECT_OUT, command, 
                                              parser->tokens[parser->position].value);
                parser->position++;
            }
        } else if (current->type == TOKEN_REDIRECT_APPEND) {
            parser->position++;
            if (parser->position < parser->count && 
                parser->tokens[parser->position].type == TOKEN_WORD) {
                command = create_redirect_node(NODE_REDIRECT_APPEND, command, 
                                              parser->tokens[parser->position].value);
                parser->position++;
            }
        } else {
            break;
        }
    }
    
    return command;
}

// Parse pipeline (e.g., "cmd1 | cmd2 | cmd3")
ASTNode* parse_pipeline(Parser *parser) {
    ASTNode *left = parse_simple_command(parser);
    
    while (parser->position < parser->count && 
           parser->tokens[parser->position].type == TOKEN_PIPE) {
        parser->position++;  // Skip pipe token
        ASTNode *right = parse_simple_command(parser);
        left = create_pipe_node(left, right);
    }
    
    return left;
}

// Parse logical operations (&&, ||)
ASTNode* parse_logical(Parser *parser) {
    ASTNode *left = parse_pipeline(parser);
    
    while (parser->position < parser->count) {
        Token *current = &parser->tokens[parser->position];
        
        if (current->type == TOKEN_AND) {
            parser->position++;
            ASTNode *right = parse_pipeline(parser);
            left = create_and_node(left, right);
        } else if (current->type == TOKEN_OR) {
            parser->position++;
            ASTNode *right = parse_pipeline(parser);
            left = create_or_node(left, right);
        } else {
            break;
        }
    }
    
    return left;
}

// Parse sequences (e.g., "cmd1 ; cmd2")
ASTNode* parse_sequence(Parser *parser) {
    ASTNode *left = parse_logical(parser);
    
    while (parser->position < parser->count && 
           parser->tokens[parser->position].type == TOKEN_SEMICOLON) {
        parser->position++;  // Skip semicolon
        ASTNode *right = parse_logical(parser);
        left = create_sequence_node(left, right);
    }
    
    return left;
}

// Main parsing function
ASTNode* parse(Token *tokens, int count) {
    Parser parser = {
        .tokens = tokens,
        .position = 0,
        .count = count
    };
    
    return parse_sequence(&parser);
}

// ============================================================================
// 4. EXECUTION - Walking the AST and executing commands
// ============================================================================

int execute_ast(ASTNode *node);

// Execute a simple command
int execute_command(ASTNode *node) {
    pid_t pid = fork();
    
    if (pid == 0) {
        // Child process
        execvp(node->data.command.argv[0], node->data.command.argv);
        perror("execvp failed");
        exit(1);
    } else if (pid > 0) {
        // Parent process
        int status;
        waitpid(pid, &status, 0);
        return WIFEXITED(status) ? WEXITSTATUS(status) : -1;
    } else {
        perror("fork failed");
        return -1;
    }
}

// Execute a pipeline
int execute_pipe(ASTNode *node) {
    int pipefd[2];
    pid_t pid1, pid2;
    
    if (pipe(pipefd) == -1) {
        perror("pipe failed");
        return -1;
    }
    
    pid1 = fork();
    if (pid1 == 0) {
        // First child - left command
        close(pipefd[0]);  // Close read end
        dup2(pipefd[1], STDOUT_FILENO);  // Redirect stdout to pipe
        close(pipefd[1]);
        
        exit(execute_ast(node->data.binary.left));
    }
    
    pid2 = fork();
    if (pid2 == 0) {
        // Second child - right command
        close(pipefd[1]);  // Close write end
        dup2(pipefd[0], STDIN_FILENO);  // Redirect stdin from pipe
        close(pipefd[0]);
        
        exit(execute_ast(node->data.binary.right));
    }
    
    // Parent closes both ends and waits
    close(pipefd[0]);
    close(pipefd[1]);
    
    int status1, status2;
    waitpid(pid1, &status1, 0);
    waitpid(pid2, &status2, 0);
    
    return WIFEXITED(status2) ? WEXITSTATUS(status2) : -1;
}

// Execute with redirection
int execute_redirect(ASTNode *node) {
    pid_t pid = fork();
    
    if (pid == 0) {
        // Child process
        int fd = open(node->data.redirect.filename, 
                     node->data.redirect.flags, 0644);
        if (fd == -1) {
            perror("open failed");
            exit(1);
        }
        
        dup2(fd, node->data.redirect.fd);
        close(fd);
        
        exit(execute_ast(node->data.redirect.command));
    } else if (pid > 0) {
        // Parent waits
        int status;
        waitpid(pid, &status, 0);
        return WIFEXITED(status) ? WEXITSTATUS(status) : -1;
    } else {
        perror("fork failed");
        return -1;
    }
}

// Execute sequence
int execute_sequence(ASTNode *node) {
    execute_ast(node->data.binary.left);
    return execute_ast(node->data.binary.right);
}

// Execute logical AND
int execute_and(ASTNode *node) {
    int result = execute_ast(node->data.binary.left);
    if (result == 0) {
        return execute_ast(node->data.binary.right);
    }
    return result;
}

// Execute logical OR
int execute_or(ASTNode *node) {
    int result = execute_ast(node->data.binary.left);
    if (result != 0) {
        return execute_ast(node->data.binary.right);
    }
    return result;
}

// Main execution dispatcher
int execute_ast(ASTNode *node) {
    if (node == NULL) {
        return -1;
    }
    
    switch (node->type) {
        case NODE_COMMAND:
            return execute_command(node);
        case NODE_PIPE:
            return execute_pipe(node);
        case NODE_REDIRECT_IN:
        case NODE_REDIRECT_OUT:
        case NODE_REDIRECT_APPEND:
            return execute_redirect(node);
        case NODE_SEQUENCE:
            return execute_sequence(node);
        case NODE_AND:
            return execute_and(node);
        case NODE_OR:
            return execute_or(node);
        default:
            fprintf(stderr, "Unknown node type: %d\n", node->type);
            return -1;
    }
}

// ============================================================================
// 5. MEMORY MANAGEMENT - Cleanup
// ============================================================================

void free_ast(ASTNode *node) {
    if (node == NULL) {
        return;
    }
    
    switch (node->type) {
        case NODE_COMMAND:
            for (int i = 0; i < node->data.command.argc; i++) {
                free(node->data.command.argv[i]);
            }
            free(node->data.command.argv);
            break;
            
        case NODE_PIPE:
        case NODE_SEQUENCE:
        case NODE_AND:
        case NODE_OR:
            free_ast(node->data.binary.left);
            free_ast(node->data.binary.right);
            break;
            
        case NODE_REDIRECT_IN:
        case NODE_REDIRECT_OUT:
        case NODE_REDIRECT_APPEND:
            free_ast(node->data.redirect.command);
            free(node->data.redirect.filename);
            break;
            
        case NODE_BACKGROUND:
            free_ast(node->data.background.command);
            break;
    }
    
    free(node);
}

// ============================================================================
// 6. HELPER FUNCTIONS FOR TESTING
// ============================================================================

void print_ast(ASTNode *node, int depth) {
    if (node == NULL) {
        return;
    }
    
    for (int i = 0; i < depth; i++) {
        printf("  ");
    }
    
    switch (node->type) {
        case NODE_COMMAND:
            printf("COMMAND: ");
            for (int i = 0; i < node->data.command.argc; i++) {
                printf("%s ", node->data.command.argv[i]);
            }
            printf("\n");
            break;
            
        case NODE_PIPE:
            printf("PIPE\n");
            print_ast(node->data.binary.left, depth + 1);
            print_ast(node->data.binary.right, depth + 1);
            break;
            
        case NODE_REDIRECT_OUT:
            printf("REDIRECT_OUT: %s\n", node->data.redirect.filename);
            print_ast(node->data.redirect.command, depth + 1);
            break;
            
        case NODE_SEQUENCE:
            printf("SEQUENCE\n");
            print_ast(node->data.binary.left, depth + 1);
            print_ast(node->data.binary.right, depth + 1);
            break;
            
        case NODE_AND:
            printf("AND\n");
            print_ast(node->data.binary.left, depth + 1);
            print_ast(node->data.binary.right, depth + 1);
            break;
            
        case NODE_OR:
            printf("OR\n");
            print_ast(node->data.binary.left, depth + 1);
            print_ast(node->data.binary.right, depth + 1);
            break;
            
        default:
            printf("UNKNOWN\n");
    }
}

// ============================================================================
// 7. EXAMPLE TOKENIZER (Simple for demonstration)
// ============================================================================

Token* tokenize(char *input, int *count) {
    Token *tokens = malloc(sizeof(Token) * 256);
    *count = 0;
    char *token = strtok(input, " \t\n");
    
    while (token != NULL) {
        if (strcmp(token, "|") == 0) {
            tokens[*count].type = TOKEN_PIPE;
            tokens[*count].value = strdup(token);
        } else if (strcmp(token, "<") == 0) {
            tokens[*count].type = TOKEN_REDIRECT_IN;
            tokens[*count].value = strdup(token);
        } else if (strcmp(token, ">") == 0) {
            tokens[*count].type = TOKEN_REDIRECT_OUT;
            tokens[*count].value = strdup(token);
        } else if (strcmp(token, ">>") == 0) {
            tokens[*count].type = TOKEN_REDIRECT_APPEND;
            tokens[*count].value = strdup(token);
        } else if (strcmp(token, ";") == 0) {
            tokens[*count].type = TOKEN_SEMICOLON;
            tokens[*count].value = strdup(token);
        } else if (strcmp(token, "&&") == 0) {
            tokens[*count].type = TOKEN_AND;
            tokens[*count].value = strdup(token);
        } else if (strcmp(token, "||") == 0) {
            tokens[*count].type = TOKEN_OR;
            tokens[*count].value = strdup(token);
        } else if (strcmp(token, "&") == 0) {
            tokens[*count].type = TOKEN_BACKGROUND;
            tokens[*count].value = strdup(token);
        } else {
            tokens[*count].type = TOKEN_WORD;
            tokens[*count].value = strdup(token);
        }
        
        (*count)++;
        token = strtok(NULL, " \t\n");
    }
    
    tokens[*count].tNype = TOKEN_EOF;
    tokens[*count].value = NULL;
    
    return tokens;
}

void free_tokens(Token *tokens, int count) {
    for (int i = 0; i < count; i++) {
        if (tokens[i].value != NULL) {
            free(tokens[i].value);
        }
    }
    free(tokens);
}

// ============================================================================
// 8. MAIN - Example Usage
// ============================================================================

int main() {
    // Example 1: Simple command with pipe
    char input1[] = "ls -la | grep test | wc -l";
    printf("Parsing: %s\n", input1);
    
    int count;
    Token *tokens = tokenize(input1, &count);
    ASTNode *ast = parse(tokens, count);
    
    printf("\nAST Structure:\n");
    print_ast(ast, 0);
    
    printf("\nExecuting...\n");
    int result = execute_ast(ast);
    printf("Result: %d\n", result);
    
    free_ast(ast);
    free_tokens(tokens, count);
    
    // Example 2: Command with redirection
    char input2[] = "echo Hello World > output.txt";
    printf("\n\nParsing: %s\n", input2);
    
    tokens = tokenize(input2, &count);
    ast = parse(tokens, count);
    
    printf("\nAST Structure:\n");
    print_ast(ast, 0);
    
    printf("\nExecuting...\n");
    result = execute_ast(ast);
    printf("Result: %d\n", result);
    
    free_ast(ast);
    free_tokens(tokens, count);
    
    // Example 3: Logical operations
    char input3[] = "test -f output.txt && echo File exists || echo File not found";
    printf("\n\nParsing: %s\n", input3);
    
    tokens = tokenize(input3, &count);
    ast = parse(tokens, count);
    
    printf("\nAST Structure:\n");
    print_ast(ast, 0);
    
    printf("\nExecuting...\n");
    result = execute_ast(ast);
    printf("Result: %d\n", result);
    
    free_ast(ast);
    free_tokens(tokens, count);
    
    return 0;
}
